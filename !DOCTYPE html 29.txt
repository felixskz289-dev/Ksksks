  <div class="note">📌 <b>Тема:</b> национальность и страны. Ключевая фраза: <b>我是 + страна + 人</b> — «я … (национальность)». Нажимайте на карточки, чтобы перевернуть, и на 🔊, чтобы услышать произношение.</div>

  <h3 class="sub">📝 生词 Новые слова</h3>
  <div class="grid" id="cardsNew"></div>

  <h3 class="sub">🧩 Грамматика</h3>
  <div class="gram">
    <p><b>1. 的 (de)</b> — показатель родительного падежа: <b>我的中文老师</b> — «мой учитель китайского». Структура: <i>прилагательное/местоимение + 的 + существительное</i>.</p>
    <p><b>2. 哪国人 (nǎ guó rén)</b> — «гражданин какой страны?»: <b>你女朋友是哪国人？</b> — «Твоя девушка из какой страны?»</p>
    <p><b>3. 也 (yě)</b> — «тоже»: <b>她也是泰国人。</b> — «Она тоже таиландка».</p>
    <p><b>4. 这 (zhè)</b> — «это» + 是: <b>这是谁？</b> — «Кто это?». Ответ: <b>这是我女朋友。</b></p>
    <p><b>5. Китайские имена</b>: фамилия + имя (姓 + 名), например 李文 = Ли (фамилия) + Вэнь (имя).</p>

function speak(text, ev){
  if(ev){ev.stopPropagation();}
  try{
    const u = new SpeechSynthesisUtterance(text);
    u.lang="zh-CN"; u.rate=0.85;
    speechSynthesis.cancel(); speechSynthesis.speak(u);
  }catch(e){}
}

/* ---------- QUIZ ---------- */
let qList=[], qIdx=0, qPts=0;
function shuffle(a){for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}return a;}
function startQuiz(){
  qList = shuffle(ALL.slice()).slice(0,10);
  qIdx=0; qPts=0;
  document.getElementById("quizPts").textContent="Очки: 0";
  nextQ();
}
function nextQ(){
  if(qIdx>=qList.length){
    document.getElementById("quizScore").textContent="Готово!";
    document.getElementById("quizBox").innerHTML=`<div style="text-align:center;padding:20px;font-size:1.2rem;">🏆 Результат: <b>${qPts} / ${qList.length}</b><br><span style="color:var(--muted);font-size:.9rem;">${qPts>=8?"Отлично! Иероглифы закреплены. 💪":qPts>=5?"Неплохо! Повтори карточки и попробуй ещё. 📚":"Стоит ещё потренироваться — загляни в разделы уроков. 🔄"}</span></div>`;
    document.getElementById("quizFb").textContent="";
    return;
  }
  const w=qList[qIdx];
  const opts=shuffle([w].concat(shuffle(ALL.filter(x=>x!==w)).slice(0,3)));
  document.getElementById("quizScore").textContent=`Вопрос ${qIdx+1} из ${qList.length}`;
  document.getElementById("quizFb").textContent="";
  document.getElementById("quizFb").className="feedback";
  document.getElementById("quizBox").innerHTML=`
    <div class="q-hz">${w.hz} <button class="speak" style="position:static;" onclick="speak('${w.hz}')">🔊</button></div>
    <div class="opts">${opts.map((o,i)=>`<button class="opt" onclick="answerQ(this,${i},${qIdx})"><b style="color:var(--red);">${o.py}</b> — ${o.tr}</button>`).join("")}</div>`;
  window._opts=opts;
}
function answerQ(btn,i,qi){
  if(qi!==qIdx) return;
  const w=qList[qIdx];
  const btns=document.querySelectorAll("#quizBox .opt");
  btns.forEach(b=>b.disabled=true);
  if(window._opts[i]===w){
    btn.classList.add("ok"); qPts++;
    document.getElementById("quizFb").textContent="✅ Верно!"; document.getElementById("quizFb").className="feedback ok";
  }else{
    btn.classList.add("bad");
    btns[[...btns].findIndex(b=>window._opts[[...btns].indexOf(b)]===w)]?.classList.add("ok");
    window._opts.forEach((o,j)=>{if(o===w) btns[j].classList.add("ok");});
    document.getElementById("quizFb").textContent=`❌ Правильный ответ: ${w.py} — ${w.tr}`; document.getElementById("quizFb").className="feedback bad";
  }
  document.getElementById("quizPts").textContent="Очки: "+qPts;
  qIdx++;
  setTimeout(nextQ, 1600);
}
startQuiz();

/* ---------- MATCHING ---------- */
let mPairs=[], mSelL=null, mSelR=null, mDone=0;
function startMatch(){
  mPairs = shuffle(ALL.slice()).slice(0,8);
  mSelL=null; mSelR=null; mDone=0;
  document.getElementById("matchScore").textContent="Совпадений: 0 / 8";
  document.getElementById("matchFb").textContent="";
  const L=shuffle(mPairs.map((w,i)=>({t:w.hz,i})));
  const R=shuffle(mPairs.map((w,i)=>({t:w.py,i})));
  document.getElementById("matchL").innerHTML=L.map(x=>`<div class="mitem" data-i="${x.i}" onclick="mPick(this,'L')">${x.t}</div>`).join("");
  document.getElementById("matchR").innerHTML=R.map(x=>`<div class="mitem" data-i="${x.i}" onclick="mPick(this,'R')">${x.t}</div>`).join("");
}
function mPick(el,side){
  if(el.classList.contains("done")) return;
  const prev = side==="L" ? mSelL : mSelR;
  if(prev) prev.classList.remove("sel");
  el.classList.add("sel");
  if(side==="L") mSelL=el; else mSelR=el;
  if(mSelL && mSelR){
    if(mSelL.dataset.i===mSelR.dataset.i){
      mSelL.classList.add("done"); mSelR.classList.add("done");
      mDone++;
      document.getElementById("matchScore").textContent=`Совпадений: ${mDone} / 8`;
      document.getElementById("matchFb").textContent="✅ Пара найдена!"; document.getElementById("matchFb").className="feedback ok";
      if(mDone===8){document.getElementById("matchFb").textContent="🏆 Все пары найдены!"; speak("很好！");}
    }else{
      document.getElementById("matchFb").textContent="❌ Не совпадает, попробуйте ещё."; document.getElementById("matchFb").className="feedback bad";
    }
    mSelL.classList.remove("sel"); mSelR.classList.remove("sel");
    mSelL=null; mSelR=null;
  }
}
startMatch();

/* ---------- SENTENCE BUILDER ---------- */
let curSent=null, picked=[];
function startBuilder(){
  curSent = SENTENCES[Math.floor(Math.random()*SENTENCES.length)];
  picked=[];
  document.getElementById("buildFb").textContent="";
  document.getElementById("buildScore").textContent="Собери: «"+curSent.tr+"»";
  const chips = shuffle(curSent.ans.map((t,i)=>({t,i})));
  document.getElementById("chipBar").innerHTML = chips.map((c,idx)=>`<button class="chip" id="chip${idx}" data-t="${c.t.replace(/"/g,"&quot;")}" onclick="pickChip(${idx})">${c.t}</button>`).join("");
  renderSlots();
}
function pickChip(idx){
  const el=document.getElementById("chip"+idx);
  if(el.classList.contains("used")) return;
  el.classList.add("used");
  picked.push({t:el.dataset.t, idx});
  renderSlots();
}
function renderSlots(){
  const bar=document.getElementById("slotBar");
  if(picked.length===0){bar.innerHTML='<span style="color:var(--muted);font-size:.9rem;">Нажимайте на слова ниже, чтобы собрать предложение…</span>';return;}
  bar.innerHTML = picked.map((p,k)=>`<span class="slot-chip" onclick="unpick(${k})">${p.t}</span>`).join("");
}
function unpick(k){
  const p=picked.splice(k,1)[0];
  document.getElementById("chip"+p.idx).classList.remove("used");
  renderSlots();
  document.getElementById("buildFb").textContent="";
}
function checkBuild(){
  if(!curSent) return;
  const got=picked.map(p=>p.t).join("|");
  const want=curSent.ans.join("|");
  const fb=document.getElementById("buildFb");
  if(got===want){
    fb.textContent=`✅ Верно! ${curSent.py}`; fb.className="feedback ok"; speak(curSent.ans.join(""));
  }else{
    fb.textContent=`❌ Пока не так. Подсказка: ${curSent.py}`; fb.className="feedback bad";
  }
}
startBuilder();

function checkFill(){
  let ok=0, total=FILL.length;
  FILL.forEach((f,i)=>{
    const sel=document.getElementById("fill"+i);
    sel.style.borderColor = (+sel.value===f.ans) ? "var(--ok)" : "var(--bad)";
    if(+sel.value===f.ans) ok++;
  });
  const fb=document.getElementById("fillFb");
  fb.textContent = ok===total ? "🏆 Отлично! Все пропуски верны." : `Верно ${ok} из ${total}. Исправь выделенные поля.`;
  fb.className = "feedback " + (ok===total?"ok":"bad");
}
renderFill();

  }

}
